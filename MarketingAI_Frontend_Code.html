2. Authentication — AuthContext

All Firebase authentication state is managed through a React context. Every page behind the dashboard is protected by this context. The useAuth() hook exposes the user object and loading state to any component.

context/AuthContext.tsx
"use client";
import { createContext, useContext, useEffect, useState } from "react";
import { onAuthStateChanged, signOut, User } from "firebase/auth";
import { auth } from "@/lib/firebase";

interface AuthContextType {
  user: User | null;
  loading: boolean;
  logout: () => Promise<void>;
}

const AuthContext = createContext<AuthContextType>({
  user: null, loading: true, logout: async () => {}
});

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Firebase listener — updates user state on login/logout
    const unsubscribe = onAuthStateChanged(auth, (firebaseUser) => {
      setUser(firebaseUser);
      setLoading(false);
    });
    return unsubscribe; // Cleanup listener on unmount
  }, []);

  const logout = async () => { await signOut(auth); };

  return (
    <AuthContext.Provider value={{ user, loading, logout }}>
      {!loading && children}
    </AuthContext.Provider>
  );
}

// Hook used by every dashboard component
export const useAuth = () => useContext(AuthContext);
3. Dashboard Layout + Auto-Publisher

The layout.tsx wraps every dashboard page. It renders the sidebar navigation, handles the notification bell, and runs the auto-publisher polling loop — checking for due scheduled posts every 60 seconds as long as a user is logged in.

app/dashboard/layout.tsx — Auto-Publisher Polling
"use client";
import { useEffect } from "react";
import { useAuth } from "@/context/AuthContext";
import { useRouter } from "next/navigation";

export default function DashboardLayout({ children }) {
  const { user, loading } = useAuth();
  const router = useRouter();

  // Redirect unauthenticated users
  useEffect(() => {
    if (!loading && !user) router.push("/auth");
  }, [user, loading]);

  // ──────────────────────────────────────────────────
  // AUTO-PUBLISHER: polls cron endpoint every 60s
  // Triggers publishing of scheduled posts silently
  // ──────────────────────────────────────────────────
  useEffect(() => {
    if (!user) return;
    // Fire immediately when dashboard opens
    fetch("/api/cron/publish-scheduled").catch(() => {});
    // Then repeat every 60 seconds
    const timer = setInterval(() =>
      fetch("/api/cron/publish-scheduled").catch(() => {}),
      60_000
    );
    return () => clearInterval(timer);
  }, [user]);

  return (
    <div className="flex min-h-screen bg-background">
      <Sidebar />
      <main className="flex-1 overflow-auto">{children}</main>
    </div>
  );
}

4. Create Content Page

The content creation workspace allows users to describe a topic, select a platform and tone, then generate an AI caption. It also includes Pexels image search, a Tone Guard check, and save/publish/schedule actions.

app/dashboard/create/page.tsx — Generate Handler
const handleGenerate = async () => {
  if (!topic.trim()) { alert("Enter a topic first"); return; }
  setGenerating(true);
  try {
    const token = await user.getIdToken();
    const res = await fetch("/api/gemini/generate", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Authorization": `Bearer ${token}`
      },
      body: JSON.stringify({ topic, platform, tone, keywords })
    });

    alert("Generation failed. Please try again.");

    setGenerating(false);
  }
};

Delete Handler with Instant UI Update
const handleDelete = async (postId: string) => {
  if (!confirm("Delete this post? This cannot be undone.")) return;
  setIsDeleting(postId);
  try {
    const token = await user.getIdToken();
    const res = await fetch(`/api/posts/${postId}`, {
      method: "DELETE",
      headers: { Authorization: `Bearer ${token}` }
    });
    if (!res.ok) throw new Error((await res.json()).error);
    // Optimistic UI — remove from state immediately, no reload
    setPosts(prev => prev.filter(p => p.id !== postId));
  } finally { setIsDeleting(null); }
};

// Schedule a draft post
const handleSchedule = async (postId: string, scheduledAt: string) => {
  const token = await user.getIdToken();
  await fetch(`/api/posts/${postId}`, {
    method: "PATCH",
    headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
    body: JSON.stringify({ status: "scheduled", scheduledAt })
  });
  await fetchPosts(); // Refresh the list
};

8. Schedule Calendar Page

A monthly calendar grid where each day cell shows chips for any published or scheduled posts on that day. Clicking a chip shows post details. Users can also navigate between months.

app/dashboard/schedule/page.tsx — Calendar Grid Logic
// Generate the calendar grid for a month
const buildCalendar = (year: number, month: number) => {
  const firstDay = new Date(year, month, 1).getDay(); // 0=Sun
  const daysInMonth = new Date(year, month + 1, 0).getDate();
  const cells: (number | null)[] = [...Array(firstDay).fill(null),
    ...Array.from({ length: daysInMonth }, (_, i) => i + 1)
  ];
  return cells;
};

// Get posts for a specific day (posts are indexed by date string)
const getPostsForDay = (day: number) =>
  posts.filter(p => {
    const d = new Date(p.scheduledAt || p.publishedAt || p.createdAt);
    return d.getFullYear() === currentYear
        && d.getMonth()    === currentMonth
        && d.getDate()     === day;
  });

ImageEditor / Pexels Picker

Searches the Pexels API for stock photos matching the user's topic and renders a responsive grid for selection. The selected image URL is stored with the post.

components/visuals/ImageEditor.tsx
const searchImages = async (query: string) => {
  const res = await fetch(
    `https://api.pexels.com/v1/search?query=${query}&per_page=12`,
    { headers: { Authorization: process.env.NEXT_PUBLIC_PEXELS_API_KEY! } }
  );
  const data = await res.json();
  setImages(data.photos.map((p: any) => p.src.large));
};
