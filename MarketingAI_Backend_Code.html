2. Firebase Admin Setup

All API routes use the Firebase Admin SDK to verify JWT tokens and read/write Firestore. The SDK is initialized once using environment variable credentials.

lib/firebase-admin.ts
import { initializeApp, getApps, getApp, cert } from "firebase-admin/app";
import { getFirestore } from "firebase-admin/firestore";
import { getAuth } from "firebase-admin/auth";
import { getStorage } from "firebase-admin/storage";

// Singleton pattern — only initialize once across hot reloads
if (getApps().length === 0) {
  initializeApp({
    credential: cert({
      projectId:   process.env.NEXT_PUBLIC_FIREBASE_PROJECT_ID,
      clientEmail: process.env.FIREBASE_ADMIN_CLIENT_EMAIL,
      privateKey:  process.env.FIREBASE_ADMIN_PRIVATE_KEY
                     ?.replace(/\\n/g, "\n"), // Fix escaped newlines
    }),
    storageBucket: process.env.NEXT_PUBLIC_FIREBASE_STORAGE_BUCKET,
  });
}

export const adminDb      = getFirestore();
export const adminStorage = getStorage();
export const adminAuth    = getAuth();

Every API route authenticates requests by verifying a Firebase JWT using getAuth().verifyIdToken(token) extracted from the Authorization: Bearer <token> header.

3. Posts API
GET /api/posts — List all posts
export async function GET(req: Request) {
  const token = req.headers.get("Authorization")!.split("Bearer ")[1];
  const { uid } = await getAuth().verifyIdToken(token);

  const snapshot = await adminDb.collection("posts")
    .where("userId", "==", uid)
    .orderBy("createdAt", "desc")
    .get();

  const posts = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
  return NextResponse.json({ posts });
}
POST /api/posts — Create post/draft
export async function POST(req: Request) {
  const { uid } = await verifyToken(req);
  const body = await req.json();

  const docRef = await adminDb.collection("posts").add({
    ...body,
    userId:    uid,
    status:    body.status || "draft",
    createdAt: new Date().toISOString(),
  });

  return NextResponse.json({ id: docRef.id }, { status: 201 });
}
PATCH /api/posts/[id] — Update (schedule, edit content)
export async function PATCH(req, { params }) {
  const { id } = await params;
  const { uid } = await verifyToken(req);
  const body = await req.json();

  const docRef = adminDb.collection("posts").doc(id);
  const doc = await docRef.get();
  if (doc.data()?.userId !== uid)
    return NextResponse.json({ error: "Forbidden" }, { status: 403 });

  await docRef.update({ ...body, updatedAt: new Date().toISOString() });
  return NextResponse.json({ success: true });
}
DELETE /api/posts/[id] — Delete with ownership guard
export async function DELETE(req, { params }) {
  const { id } = await params;
  const { uid } = await verifyToken(req);

  const docRef = adminDb.collection("posts").doc(id);
  const doc = await docRef.get();
  if (!doc.exists)
    return NextResponse.json({ error: "Not found" }, { status: 404 });
  if (doc.data()?.userId !== uid)
    return NextResponse.json({ error: "Forbidden" }, { status: 403 });

  await docRef.delete();
  return NextResponse.json({ success: true });
}
4. Auto-Publisher Cron Route

Queries Firestore for all posts with status == "scheduled" and scheduledAt <= now, then calls SocialMediaManager.publishPost() for each. The dashboard polls this every 60 seconds.

app/api/cron/publish-scheduled/route.ts
export async function GET() {
  const now = new Date().toISOString();

  const snapshot = await adminDb.collection("posts")
    .where("status",      "==",  "scheduled")
    .where("scheduledAt", "<=", now)
    .get();

  if (snapshot.empty)
    return NextResponse.json({ message: "No posts due", count: 0 });

  let published = 0, failed = 0;
  for (const doc of snapshot.docs) {
    try {
      await SocialMediaManager.publishPost(doc.id, doc.data().userId);
      published++;
    } catch (e) {
      console.error(`[CRON] Failed ${doc.id}:`, e); failed++;
    }
  }
  return NextResponse.json({ published, failed });
}
SocialMediaManager — Idempotent publishPost()
static async publishPost(postId: string, userId: string) {
  const postRef = adminDb.collection("posts").doc(postId);
  const post = (await postRef.get()).data()!;

  // Idempotency — skip if already published or mid-publish
  if (["published","publishing"].includes(post.status))
    return { success: true, skipped: true };

  // Lock against race conditions
  await postRef.update({ status: "publishing" });

  // Route to correct platform connector
  const connector = await getConnector(post.platform, userId);
  const platformPostId = await connector.publish(post.content, post.imageUrl);

  await postRef.update({
    status: "published", platformPostId,
    publishedAt: new Date().toISOString()
  });
  return { success: true, platformPostId };
}
5. Analytics Sync + Follower Snapshots
app/api/analytics/sync/route.ts
export async function POST(req: Request) {
  const { uid } = await verifyToken(req);

  // 1. Sync post metrics from platform APIs
  const posts = await adminDb.collection("posts")
    .where("userId", "==", uid)
    .where("status", "==", "published").get();

  await Promise.allSettled(
    posts.docs.map(doc => SocialMediaManager.syncPostMetrics(doc.id, uid))
  );

  // 2. Save follower snapshots for each connected platform
  const connections = await adminDb
    .collection("users").doc(uid)
    .collection("connections").get();

  for (const conn of connections.docs) {
    const platform  = conn.id;       // "instagram" | "linkedin"
    const data      = conn.data();
    let   count     = 0;

    if (platform === "instagram") {
      const { InstagramConnector } = await import("@/lib/social/instagram");
      count = await new InstagramConnector(data.igAccountId, data.accessToken)
                .getFollowerCount();
    } else if (platform === "linkedin") {
      const { LinkedInConnector } = await import("@/lib/social/linkedin");
      count = await new LinkedInConnector(data.linkedinId, data.accessToken)
                .getFollowerCount();
    }
    if (count > 0) await saveFollowerSnapshot(uid, count, platform);
  }
  return NextResponse.json({ success: true });
}
lib/agent/follower-snapshot.ts — Daily Deduplication
export async function saveFollowerSnapshot(
  userId: string, followerCount: number, platform: string
) {
  const today = new Date().toISOString().slice(0, 10); // "YYYY-MM-DD"

  // Check for existing snapshot today (deduplication)
  const existing = await adminDb.collection("follower_snapshots")
    .where("userId",   "==", userId)
    .where("platform", "==", platform)
    .where("date",     "==", today)
    .limit(1).get();

  if (!existing.empty) {
    // Update today's existing record
    await existing.docs[0].ref.update({ followerCount });
  } else {
    // Create new daily snapshot
    await adminDb.collection("follower_snapshots").add({
      userId, platform, followerCount, date: today,
      recordedAt: new Date().toISOString()
    });
  }
}
6. Strategy Agent Pipeline

Three sequential modules: Observer enriches posts, Analyzer detects patterns, Planner calls Gemini to generate strategy.

Module 1 — Observer (lib/agent/observer.ts)
export async function observeAllPosts(uid: string) {
  const posts = await getPublishedPosts(uid);

  for (const post of posts) {
    const m = post.metrics || {};

    // Weighted engagement score
    const engagementScore =
      (0.5 * (m.likes    || 0)) +
      (0.3 * (m.comments || 0)) +
      (0.2 * (m.shares   || 0));

    const date         = new Date(post.publishedAt);
    const postHour     = date.getHours();
    const postDayOfWk  = date.getDay();
    const hashtags     = (post.content.match(/#[\w]+/g) || []);
    const category     = categorize(post.content);

    await adminDb.collection("posts").doc(post.id).update({
      engagementScore, postHour, postDayOfWeek: postDayOfWk,
      hashtags, contentCategory: category
    });
  }
}
Module 2 — Analyzer (lib/agent/analyzer.ts)
export async function analyzePosts(uid: string) {
  const posts = await getEnrichedPosts(uid);

  // Build hourly engagement array [0..23]
  const hourlyEng = new Array(24).fill(0);
  const hourCounts = new Array(24).fill(0);
  posts.forEach(p => {
    hourlyEng[p.postHour]   += p.engagementScore;
    hourCounts[p.postHour]  += 1;
  });
  const avgHourly = hourlyEng.map((s, i) => hourCounts[i] ? s/hourCounts[i] : 0);
  const bestHour  = avgHourly.indexOf(Math.max(...avgHourly));

  // Platform + tone breakdown
  const byPlatform = groupAvg(posts, "platform");
  const byTone     = groupAvg(posts, "tone");

  await adminDb.collection("agent_insights").doc(uid).set({
    bestHour, avgHourly,
    bestPlatform: maxKey(byPlatform),
    bestTone:     maxKey(byTone),
    platformBreakdown: byPlatform,
    toneBreakdown:     byTone,
    updatedAt:    new Date().toISOString()
  });
}
Module 3 — Planner (lib/agent/planner.ts)
export async function generateStrategy(uid: string) {
  const insights = (await adminDb.collection("agent_insights").doc(uid).get()).data()!;
  const profile  = (await adminDb.collection("users").doc(uid).get()).data()!;

  const prompt = `You are an expert social media strategist.
Business: ${profile.businessName} (${profile.industry})
Insights:
- Best posting hour: ${insights.bestHour}:00
- Best platform: ${insights.bestPlatform}
- Best tone: ${insights.bestTone}
- Top hashtags: ${insights.topHashtags?.join(", ")}
Generate a JSON strategy with: weeklyPostTarget, recommendedPostTimes,
contentMix (object), recommendedTone, reasoning (array of strings).`;

  const gemini   = new GeminiClient({ apiKey: process.env.GEMINI_API_KEY! });
  const raw      = await gemini.generateText(prompt);
  const strategy = JSON.parse(raw.replace(/```json|```/g, "").trim());

  await adminDb.collection("agent_strategy").doc(uid).set({
    ...strategy, generatedAt: new Date().toISOString(), status: "active"
  });
}
7. Gemini AI Client

Handles all AI text generation with a 3-model cascading fallback. If all Gemini models fail, it tries HuggingFace then Pollinations.ai.

lib/gemini-client.ts
async generateText(prompt: string): Promise<string> {
  // Try each active Gemini model in order
  for (const model of [
    "gemini-2.5-flash",      // Newest — 5 RPM free
    "gemini-2.0-flash",      // Stable — 15 RPM, daily quota
    "gemini-2.0-flash-lite", // Lighter — separate quota pool
  ]) {
    try { return await this.callGeminiWithModel(model, prompt); }
    catch (e: any) { console.warn(`[Gemini] ${model} failed`); }
  }
  // Fallback: HuggingFace Zephyr-7B
  try { return await this.generateTextHF(prompt, hfKey); }
  catch (e) { console.warn("HuggingFace failed"); }
  // Final: Pollinations.ai (no key needed)
  return await this.generateTextFallback(prompt);
}

private async callGeminiWithModel(model: string, prompt: string) {
  const res = await fetch(
    `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${this.apiKey}`,
    { method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] }) }
  );
  if (!res.ok) throw new Error(`Model ${model} failed (${res.status})`);
  const data = await res.json();
  return data.candidates?.[0]?.content?.parts?.[0]?.text || "";
}
8. Social Connectors
InstagramConnector — lib/social/instagram.ts
export class InstagramConnector implements SocialConnector {
  constructor(
    private igAccountId: string,
    private accessToken: string
  ) {}

  async publish(content: string, imageUrl: string): Promise<string> {
    // Step 1: Create media container
    const container = await fetch(
      `https://graph.facebook.com/v18.0/${this.igAccountId}/media`,
      { method: "POST", body: JSON.stringify({
        image_url: imageUrl, caption: content, access_token: this.accessToken
      })}
    );
    const { id: creationId } = await container.json();

    // Step 2: Poll until container ready
    let status = "IN_PROGRESS";
    while (status !== "FINISHED") {
      await await new Promise(r => setTimeout(r, 2000));
      const s = await (await fetch(
        `https://graph.facebook.com/v18.0/${creationId}?fields=status_code&access_token=${this.accessToken}`
      )).json();
      status = s.status_code;
    }

    // Step 3: Publish from container
    const pub = await (await fetch(
      `https://graph.facebook.com/v18.0/${this.igAccountId}/media_publish`,
      { method: "POST", body: JSON.stringify({
        creation_id: creationId, access_token: this.accessToken
      })}
    )).json();
    return pub.id; // IG Media ID
  }

  async getFollowerCount(): Promise<number> {
    const res = await fetch(
      `https://graph.facebook.com/v18.0/${this.igAccountId}?fields=followers_count&access_token=${this.accessToken}`
    );
    const data = await res.json();
    return data.followers_count || 0;
  }
}
LinkedInConnector — lib/social/linkedin.ts
async publish(content: string): Promise<string> {
  const res = await fetch("https://api.linkedin.com/v2/ugcPosts", {
    method: "POST",
    headers: { Authorization: `Bearer ${this.accessToken}`,
               "Content-Type": "application/json",
               "X-Restli-Protocol-Version": "2.0.0" },
    body: JSON.stringify({
      author: `urn:li:person:${this.memberId}`,
      lifecycleState: "PUBLISHED",
      specificContent: { "com.linkedin.ugc.ShareContent": {
        shareCommentary:  { text: content },
        shareMediaCategory: "NONE"
      }},
      visibility: { "com.linkedin.ugc.MemberNetworkVisibility": "PUBLIC" }
    })
  });
  const data = await res.json();
  return data.id;
}

// Step 1: Redirect user to Facebook OAuth dialog
export async function GET() {
  const url = `https://www.facebook.com/v18.0/dialog/oauth?` +
    new URLSearchParams({
      client_id:    process.env.FACEBOOK_APP_ID!,
      redirect_uri: process.env.INSTAGRAM_REDIRECT_URI!,
      scope:        "instagram_basic,instagram_content_publish,pages_show_list"
    });
  return NextResponse.redirect(url);
}

// Step 2: Exchange code for token, save to Firestore
export async function GET(req: Request) {
  const code = new URL(req.url).searchParams.get("code")!;

  // Exchange code for short-lived token
  const tokenRes = await fetch(
    `https://graph.facebook.com/v18.0/oauth/access_token?` +
    new URLSearchParams({ client_id, client_secret, redirect_uri, code })
  );
  const { access_token } = await tokenRes.json();

  // Exchange for long-lived token (60 days)
  const longLived = await exchangeForLongLivedToken(access_token);

  // Save credentials to user's connections subcollection
  await adminDb.collection("users").doc(uid)
    .collection("connections").doc("instagram")
    .set({ accessToken: longLived, igAccountId, connectedAt: new Date().toISOString() });

  return NextResponse.redirect("/dashboard/settings?connected=instagram");
}
